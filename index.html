Hubungkan ke Google Spreadsheet

URL Google Apps Script Web App sekarang dibaca langsung dari index.html. Untuk mengganti koneksi, edit nilai GOOGLE_APPS_SCRIPT_URL pada bagian JavaScript di index.html.

Catatan Integrasi:

Google Sheet adalah satu-satunya sumber data siswa. Isi URL deployment /exec pada konstanta GOOGLE_APPS_SCRIPT_URL di index.html.

Panduan Menghubungkan Google Spreadsheet

  1. Buat Google Spreadsheet Baru

    Tidak perlu membuat header manual. Kode Apps Script akan otomatis membuat tab MutasiSiswa, Akun, dan ActivityLog beserta header dan akun awal.

  2. Buka Apps Script

    Di Google Sheets, klik menu Ekstensi -> Apps Script.

  3. Paste Kode Berikut ke Apps Script: 
    const DATA_SHEET = 'MutasiSiswa';
const ACCOUNT_SHEET = 'Akun';
const ACTIVITY_SHEET = 'ActivityLog';

const STUDENT_HEADERS = [
  'id', 'jenis_mutasi', 'status_operator', 'nama_siswa',
  'nisn', 'nis', 'asal_sekolah', 'tgl_diterima', 'kelas',
  'alasan_mutasi', 'no_hp_ortu', 'no_hp_siswa', 'jenis_kelamin'
];

const ACCOUNT_HEADERS = ['username', 'password', 'role', 'nama_lengkap', 'aktif'];
const ACTIVITY_HEADERS = [
  'id', 'timestamp', 'username', 'nama_lengkap', 'role',
  'action', 'student_id', 'student_name', 'jenis_mutasi', 'description'
];

function doGet(e) {
  try {
    setupSheets_();

    const action = e && e.parameter && e.parameter.action
      ? String(e.parameter.action)
      : 'getStudents';

    if (action === 'getAccounts') {
      return json_(readObjects_(getSheet_(ACCOUNT_SHEET)));
    }

    if (action === 'getActivities') {
      const rows = readObjects_(getSheet_(ACTIVITY_SHEET));
      rows.sort((a, b) => String(b.timestamp || '').localeCompare(String(a.timestamp || '')));
      return json_(rows.slice(0, 100));
    }

    if (action === 'ping') {
      return json_({ status: 'ok', source: 'Google Sheets' });
    }

    // getStudents: Google Sheet adalah satu-satunya sumber data siswa.
    // Pembersihan duplikat dilindungi lock agar tidak bertabrakan dengan proses tulis.
    const readLock = LockService.getScriptLock();
    try {
      readLock.waitLock(10000);
      dedupeStudentSheet_();
    } finally {
      try { readLock.releaseLock(); } catch (e2) {}
    }
    return json_(readObjects_(getSheet_(DATA_SHEET)));
  } catch (err) {
    return json_({ status: 'error', message: String(err && err.message ? err.message : err) });
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    setupSheets_();

    const body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    const action = String(body.action || '');

    if (action === 'upsertStudent') {
      const result = upsertStudent_(body.student || {});
      return json_({ status: 'success', action: action, student: result });
    }

    if (action === 'updateStatus') {
      const id = String(body.id || '').trim();
      const status = String(body.status || '').trim();
      if (!id) throw new Error('ID siswa wajib diisi.');
      if (!status) throw new Error('Status wajib diisi.');

      const row = findStudentRowById_(id);
      if (!row) throw new Error('Data siswa tidak ditemukan.');

      const sheet = getSheet_(DATA_SHEET);
      const statusCol = STUDENT_HEADERS.indexOf('status_operator') + 1;
      sheet.getRange(row, statusCol).setValue(status);
      return json_({ status: 'success', action: action, id: id });
    }

    if (action === 'deleteStudent') {
      const id = String(body.id || '').trim();
      if (!id) throw new Error('ID siswa wajib diisi.');
      const deleted = deleteStudentById_(id);
      return json_({ status: 'success', action: action, deleted: deleted });
    }

    if (action === 'importStudents') {
      const rows = Array.isArray(body.students) ? body.students : [];
      let saved = 0;
      rows.forEach(item => {
        upsertStudent_(item || {});
        saved++;
      });
      dedupeStudentSheet_();
      return json_({ status: 'success', action: action, saved: saved });
    }

    if (action === 'addActivity') {
      const activity = normalizeActivity_(body.activity || {});
      if (!activity.id) activity.id = Utilities.getUuid();
      if (!activity.timestamp) activity.timestamp = new Date().toISOString();

      const sheet = getSheet_(ACTIVITY_SHEET);
      const existing = readObjects_(sheet);
      if (!existing.some(x => String(x.id || '') === activity.id)) {
        sheet.appendRow(ACTIVITY_HEADERS.map(h => activity[h] === undefined ? '' : activity[h]));
      }
      trimActivityLog_();
      return json_({ status: 'success', action: action, activity_id: activity.id });
    }

    if (action === 'cleanupDuplicates') {
      const removed = dedupeStudentSheet_();
      return json_({ status: 'success', action: action, removed: removed });
    }

    return json_({ status: 'error', message: 'Action tidak dikenali.' });
  } catch (err) {
    return json_({ status: 'error', message: String(err && err.message ? err.message : err) });
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function setupSheets_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  let dataSheet = ss.getSheetByName(DATA_SHEET);
  if (!dataSheet) dataSheet = ss.insertSheet(DATA_SHEET);
  ensureHeaders_(dataSheet, STUDENT_HEADERS);

  let accountSheet = ss.getSheetByName(ACCOUNT_SHEET);
  if (!accountSheet) accountSheet = ss.insertSheet(ACCOUNT_SHEET);
  ensureHeaders_(accountSheet, ACCOUNT_HEADERS);

  if (accountSheet.getLastRow() === 1) {
    accountSheet.getRange(2, 1, 3, ACCOUNT_HEADERS.length).setValues([
      ['admin', 'admin123', 'ADMIN', 'Admin Utama', true],
      ['staff', 'staff123', 'STAFF', 'Staff Kesiswaan', true],
      ['operator', 'operator123', 'OPERATOR', 'Operator Dapodik', true]
    ]);
  }

  let activitySheet = ss.getSheetByName(ACTIVITY_SHEET);
  if (!activitySheet) activitySheet = ss.insertSheet(ACTIVITY_SHEET);
  ensureHeaders_(activitySheet, ACTIVITY_HEADERS);
}

function ensureHeaders_(sheet, expectedHeaders) {
  if (sheet.getLastRow() === 0) {
    sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
    return;
  }

  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const currentHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0]
    .map(v => String(v || '').trim());

  const exact = currentHeaders.length === expectedHeaders.length &&
    expectedHeaders.every((h, i) => currentHeaders[i] === h);
  if (exact) return;

  const values = sheet.getDataRange().getValues();
  const objects = values.slice(1)
    .filter(row => row.some(v => String(v).trim() !== ''))
    .map(row => {
      const obj = {};
      currentHeaders.forEach((h, i) => {
        if (h) obj[h] = row[i];
      });
      return obj;
    });

  sheet.clearContents();
  sheet.getRange(1, 1, 1, expectedHeaders.length).setValues([expectedHeaders]);
  if (objects.length) {
    const rows = objects.map(obj => expectedHeaders.map(h => obj[h] === undefined ? '' : obj[h]));
    sheet.getRange(2, 1, rows.length, expectedHeaders.length).setValues(rows);
  }
}

function normalizeStudent_(raw) {
  const s = {};
  STUDENT_HEADERS.forEach(h => {
    s[h] = raw[h] === undefined || raw[h] === null ? '' : String(raw[h]).trim();
  });

  s.id = s.id || Utilities.getUuid();
  s.jenis_mutasi = s.jenis_mutasi === 'KELUAR' ? 'KELUAR' : 'MASUK';
  s.status_operator = s.status_operator || 'Belum Diproses';

  if (!s.nama_siswa) throw new Error('Nama siswa wajib diisi.');
  if (!s.nisn) throw new Error('NISN wajib diisi.');
  if (!s.kelas) throw new Error('Kelas wajib diisi.');
  if (!s.tgl_diterima) throw new Error('Tanggal mutasi wajib diisi.');

  return s;
}

function normalizeActivity_(raw) {
  const a = {};
  ACTIVITY_HEADERS.forEach(h => {
    a[h] = raw[h] === undefined || raw[h] === null ? '' : String(raw[h]).trim();
  });
  return a;
}

function upsertStudent_(raw) {
  const student = normalizeStudent_(raw);
  const sheet = getSheet_(DATA_SHEET);
  const rows = readObjects_(sheet);

  let targetRow = 0;
  let existingId = '';
  const newKey = studentUniqueKey_(student);

  for (let i = 0; i < rows.length; i++) {
    const rowObj = rows[i];
    if (String(rowObj.id || '') === student.id) {
      targetRow = i + 2;
      existingId = String(rowObj.id || '');
      break;
    }
    if (!targetRow && newKey && studentUniqueKey_(rowObj) === newKey) {
      targetRow = i + 2;
      existingId = String(rowObj.id || '');
    }
  }

  if (existingId) student.id = existingId;
  const rowValues = STUDENT_HEADERS.map(h => student[h] === undefined ? '' : student[h]);

  if (targetRow) {
    sheet.getRange(targetRow, 1, 1, STUDENT_HEADERS.length).setValues([rowValues]);
  } else {
    sheet.appendRow(rowValues);
  }

  return student;
}

function studentUniqueKey_(student) {
  const nisn = String(student.nisn || '').trim();
  const jenis = String(student.jenis_mutasi || '').trim();
  const tanggal = normalizeDateValue_(student.tgl_diterima);
  if (!nisn || !jenis || !tanggal) return '';
  return [nisn, jenis, tanggal].join('|').toLowerCase();
}

function findStudentRowById_(id) {
  const sheet = getSheet_(DATA_SHEET);
  const values = sheet.getDataRange().getValues();
  const idCol = STUDENT_HEADERS.indexOf('id');
  for (let r = 1; r < values.length; r++) {
    if (String(values[r][idCol] || '').trim() === id) return r + 1;
  }
  return 0;
}

function deleteStudentById_(id) {
  const sheet = getSheet_(DATA_SHEET);
  const values = sheet.getDataRange().getValues();
  const idCol = STUDENT_HEADERS.indexOf('id');
  let deleted = 0;

  for (let r = values.length - 1; r >= 1; r--) {
    if (String(values[r][idCol] || '').trim() === id) {
      sheet.deleteRow(r + 1);
      deleted++;
    }
  }
  return deleted;
}

function dedupeStudentSheet_() {
  const sheet = getSheet_(DATA_SHEET);
  const rows = readObjects_(sheet);
  if (!rows.length) return 0;

  const seenIds = {};
  const seenKeys = {};
  const unique = [];
  let removed = 0;

  rows.forEach(raw => {
    const student = {};
    STUDENT_HEADERS.forEach(h => student[h] = raw[h] === undefined ? '' : raw[h]);
    let id = String(student.id || '').trim();
    if (!id) {
      id = Utilities.getUuid();
      student.id = id;
    }

    const key = studentUniqueKey_(student);
    if (seenIds[id] || (key && seenKeys[key])) {
      removed++;
      return;
    }

    seenIds[id] = true;
    if (key) seenKeys[key] = true;
    unique.push(student);
  });

  if (!removed && unique.length === rows.length && unique.every(x => String(x.id || '').trim())) {
    return 0;
  }

  sheet.clearContents();
  sheet.getRange(1, 1, 1, STUDENT_HEADERS.length).setValues([STUDENT_HEADERS]);
  if (unique.length) {
    sheet.getRange(2, 1, unique.length, STUDENT_HEADERS.length)
      .setValues(unique.map(s => STUDENT_HEADERS.map(h => s[h] === undefined ? '' : s[h])));
  }
  return removed;
}

function trimActivityLog_() {
  const sheet = getSheet_(ACTIVITY_SHEET);
  const lastRow = sheet.getLastRow();
  // Header + 500 aktivitas terbaru.
  if (lastRow > 501) {
    sheet.deleteRows(2, lastRow - 501);
  }
}

function getSheet_(name) {
  return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name);
}

function readObjects_(sheet) {
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];

  const headers = values[0].map(h => String(h).trim());
  return values.slice(1)
    .filter(row => row.some(v => String(v).trim() !== ''))
    .map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        if (!h) return;
        const value = row[i];
        obj[h] = value instanceof Date
          ? Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd')
          : value;
      });
      return obj;
    });
}

function normalizeDateValue_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  return String(value || '').split('T')[0].trim();
}

function json_(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
